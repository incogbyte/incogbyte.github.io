function safeObjCToString(obj) {
    if (!obj || obj.isNull()) {
        return null;
    }

        // Try direct toString first
        return obj.toString();
    } catch (e1) {
        try {

function readSwiftString(ptr) {
    if (!ptr || ptr.isNull()) {
        return null;
    }

        const firstWord = ptr.readU64();
